#define NUMVOICES 16

global int16_t g_audio_buffer[128];
global int16_t g_dma_buffer[128];

//pot->adsr flags

//midi->env flags
bool new_press[NUMVOICES];
bool new_release[NUMVOICES];

void ISR_audioblock () {
    //generate audio samples for entire audio buffer
    
    //1. if any pending new_press/new_release flags, deal with them
    for (int i=0; i<NUMVOICES; i++) {
        if (new_press[i])
            new_press[i] = false;
            press_voice(i);
        else if (new_release[i])
            new_release[i] = false;
            release_voice[i];
    }
    //2. increment oscs, update env every 16th cycle
        env_timer = 0;
        for (int buffer_i = 0; < 128; buffer_i++) {
            update_oscs();
            if (env_osc >= 16) {
                update_env();
                env_timer = 0;
            }
        }
        
    
}

void update_env () {
    for (int i = 0; i < NUM_VOICES; i++) {
        Voice_t *v = &g_voices[i];
        uint32_t level = 0;
        uint8_t requests = v->env_requests;
        v->env_requests = ENV_REQUEST_NONE;

        if (requests & ENV_REQUEST_ATTACK) {
        v->env_idx = 0;
        v->env_base_level = 0;
        v->env_state = ENV_STATE_ATTACK;
        }
        if (requests & ENV_REQUEST_RELEASE) {
        v->env_idx = 0;
        v->env_amp_latch = v->env_base_level;
        v->env_state = ENV_STATE_RELEASE;
        }

        switch (v->env_state) {
        case ENV_STATE_OFF:
            v->env_idx = 0;
            break;
        case ENV_STATE_ATTACK:
            level = env_a_table[v->env_idx];
            v->env_idx += v->a_inc;
            if (v->env_idx >= ENV_WAVETABLE_SIZE) {
            v->env_idx = 0;
            v->env_state = ENV_STATE_DECAY;
            }
            break;
        case ENV_STATE_DECAY: {
            uint32_t drop = UINT16_MAX - v->s_level;
            level = UINT16_MAX - ((drop * (UINT16_MAX - env_d_table[v->env_idx])) >> 16);
            v->env_idx += v->d_inc;
            if (v->env_idx >= ENV_WAVETABLE_SIZE) {
            v->env_idx = 0;
            v->env_state = ENV_STATE_SUSTAIN;
            }
            break;
        }
        case ENV_STATE_SUSTAIN:
            level = v->s_level;
            break;
        case ENV_STATE_RELEASE:
            level = ((uint32_t)v->env_amp_latch * env_r_table[v->env_idx]) >> 16;
            v->env_idx += v->r_inc;
            if (v->env_idx >= ENV_WAVETABLE_SIZE) {
            v->env_idx = 0;
            v->env_state = ENV_STATE_OFF;
            }
            break;
        }

        v->env_base_level = (uint16_t)level;
        v->env_level = (level * v->velocity) >> 7;
        // >> 7 same as /128, scaling by max velocity value
    }
}

void voice_allocation () {
    //check if MIDI ring buffer has any pending messages; if not, return
    if (g_midi_rb.head == g_midi_rb.tail)
        return;
    //dequeue the ring buffer
    MidiNote_t note = dq_midi_rb(&g_midi_rb);
    //decode note number into phase inc
    uint32_t note_phase_inc = midi_to_phase_inc[note.number];

    if (note.event == MIDI_NOTE_OFF || note.velocity == 0) {
        //search for an existing voice that's pressed and unpress it
        for (int i=0; i<NUM_VOICES; i++)
            if (g_voices[i].phase_inc == note_phase_inc && g_voices[i].pressed == true) {
                release_voice(&g_voices[i]);
                return;
            }
    }
    else if (note.event == MIDI_NOTE_ON) {
        //increment age
        
        //see if the array contains that note already
        //if so, restart the voice
        for (int i=0; i<NUM_VOICES; i++) {
            g_voices[i].age++;
            if (g_voices[i].phase_inc == note_phase_inc && g_voices[i].pressed == true) {
                press_voice (&g_voices[i], note_phase_inc, note.velocity);
            }
        }
        //if not, choose an empty voice
        uint32_t age_oldest=0;
        uint32_t idx_oldest=0;
        for (int i=0; i<NUM_VOICES; i++) {
            // Q** : does it matter if pressed == true?
            if (g_voices[i].env_state == ENV_STATE_OFF) {
                press_voice (&g_voices[i], note_phase_inc, note.velocity);
                return;
            }
                
            //keep track of oldest voice (index)
            if (g_voices[i].age >= age_oldest) {
                age_oldest = g_voices[i].age;
                idx_oldest = i;
            }
        }
        //if not, replace the oldest voice
        press_voice (&g_voices[idx_oldest], note_phase_inc, note.velocity);
    }
    //all other commands are ignored
    return;
}

int main () {
    //initalize all dma, i2s, interrupts etc

    for (;;) {
        voice_allocation();
    }
}
